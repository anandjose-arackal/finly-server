def generate_trail_body(user = nil)
  body = ""
  action_user = User.find_by(id: self.action_user_id)
  if user && action_user && user.id == action_user.id
    user_name = "You"
  else
    user_name = action_user.name rescue nil
    if user_name.nil? && action_id == TrailAction.id_for(:auto_approve)
      user_name = "Your"
    end
  end

  if self.action_id == TrailAction.id_for(:auto_approve)
    body = "#{self.parent_type} was auto approved. Initiated by #{user_name}"
    return body
  end

  if self.action_id == TrailAction.id_for(:send_to_vendor)
    body = "#{user_name} has sent the #{self.parent_type} to vendor."
    return body
  end

  if self.action_id == TrailAction.id_for(:revision)
    body = "#{user_name} has created a revision for this #{self.parent_type}."
    return body
  end

  if self.action_id == TrailAction.id_for(:vendor_confirmed)
    body = "#{user_name} marked the #{self.parent_type} that all details in this document is confirmed by vendor."
    return body
  end

  body += user_name

  case self.action_id
  when TrailAction.id_for(:submit)
      action = " submitted"
  when TrailAction.id_for(:reject)
      action = " rejected"
  when TrailAction.id_for(:create)
      action = " created"
  when TrailAction.id_for(:take_control)
      action = " took control"
  when TrailAction.id_for(:update)
      action = " updated"
  when TrailAction.id_for(:approve)
      action = " approved"
  when TrailAction.id_for(:reject)
      action = " rejected"
  when TrailAction.id_for(:retract)
      action = " retracted"
  when TrailAction.id_for(:unapprove)
      action = " unapproved"
  when TrailAction.id_for(:approve_and_forward)
      action = " approved and forwarded"
  when TrailAction.id_for(:auto_approve)
      action = ""
  when TrailAction.id_for(:credited)
      action = " credited"
  when TrailAction.id_for(:settled)
      action = " settled"
  when TrailAction.id_for(:settlement_confirmed)
      action = " confirmed settlement"
  when TrailAction.id_for(:reimbursed)
      action = " reimbursed"
  when TrailAction.id_for(:notify_credit_reimbursement)
      action = ""
  when TrailAction.id_for(:reimbursement_confirmed)
      action = " confirmed reimbursement"
  when TrailAction.id_for(:closed), TrailAction.id_for(:close)
      action = " closed"
  when TrailAction.id_for(:void)
      action = " voided"
  when TrailAction.id_for(:send_to_vendor)
      action = " has sent to vendor"
  when TrailAction.id_for(:revision)
      action = " created a revision for this"
  when TrailAction.id_for(:reopen)
      action = " reopened"
  when TrailAction.id_for(:forward)
      action = " forwarded"
  when TrailAction.id_for(:deactivate)
      action = " deactivated"
  when TrailAction.id_for(:reactivate)
      action = " reactivated"
  when TrailAction.id_for(:settle)
    action = " settled"
  else
      action = ""
  end
  if self.parent_type == 'YesBankVendor'
    doc = 'Vendor'
    body += action + " the #{doc}."    
  else
    body += action + " the #{self.parent_type}."    
  end

  body
end
