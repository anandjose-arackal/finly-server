def file_format
  if self.attachment_content_type == 'application/pdf'
    type = "pdf"
  elsif ["image/jpg", "image/jpeg", "image/png", "image/gif"].include?(self.attachment_content_type)
    type = "image"
  else
    type = ''
  end
  type
end
