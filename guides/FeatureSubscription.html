def bank_transaction_class
  case self.bank_service_provider
  when "YES"
    return YesBankTransaction
  when "ICICI"
    return IciciBankTransaction
  when "KOTAK"
    return KotakBankTransaction
  else
    return YesBankTransaction
  end
end
