def self.seed_data(company_id)
  DEFAULTS.each do |rate|
    if TdsRatesList.find_by(name: rate[:name], company_id: company_id).present?
      next
    else
      TdsRatesList.create(name: rate[:name], tds: rate[:tds], company_id: company_id)
    end
  end
end
