def self.seed_data
  if FeedbackType.count == 0
    FeedbackType::VALUES.each do |key, value|
      description = FeedbackType::DESCRIPTIONS[key]
      FeedbackType.create id: value, name: key.to_s, description: description
    end
  end
end
