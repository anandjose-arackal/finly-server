def generate_body(user = nil)
  content = ""

  action_user = User.find_by(id: self.action_user_id)
  if user and action_user and user.id == action_user.id
    user_name = "You"
  else
    user_name = action_user.name rescue nil
    if user_name.nil? and action_type_id == NotificationActionType.id_for(:notify_auto_approve)
      user_name = "Your"
    end
  end

  content += user_name

  # preposition selector
  if action_user_id.nil?
    prep = ""
  elsif action_user_id == user_id
    prep = " have"
  else
    prep = " has"
  end

  object_prep = retrieve_object_preposition

  content += prep # short for preposition
  case self.action_type_id
    when NotificationActionType.id_for(:notify_submit)
      action = " submitted"
    when NotificationActionType.id_for(:notify_approve)
      action = " approved"
    when NotificationActionType.id_for(:notify_reject)
      action = " rejected"
    when NotificationActionType.id_for(:notify_retract)
      action = " retracted"
    when NotificationActionType.id_for(:notify_unapprove)
      action = " unapproved"
    when NotificationActionType.id_for(:notify_auto_approve)
      action = ""
    when NotificationActionType.id_for(:notify_awaiting_approval)
      action = " approved"
    when NotificationActionType.id_for(:notify_credit_reimbursement)
      action = " credited a reimbursement of"
      object_prep = ""
    when NotificationActionType.id_for(:notify_update)
      action = " updated"
    when NotificationActionType.id_for(:notify_take_control)
      action = " taken control of"
    when NotificationActionType.id_for(:notify_comment)
      action = " commented on"
    when NotificationActionType.id_for(:notify_cash_withdrawal)
      action = " withdrawn an amount of"
    when NotificationActionType.id_for(:notify_cash_load)
      action = " added an amount of"
    when NotificationActionType.id_for(:notify_atm_withdrawal)
      action = " withdrawn an amount of"
    when NotificationActionType.id_for(:notify_credit_reimbursement)
      action = " credited the reimbursement"
    when NotificationActionType.id_for(:notify_credit_advance_request)
      action = " credited an amount of"
    when NotificationActionType.id_for(:notify_credit)
      action = " credited an amount of"
    when NotificationActionType.id_for(:notify_debit)
      action = " debited an amount of"
    when NotificationActionType.id_for(:notify_cash_withdrawal_against_report_reimbursement)
      action = " withdrawn funds from your cash on hand against reimbursement"
    when NotificationActionType.id_for(:notify_auto_reload_card_failure)
      content = "Auto reload of your expense card with amount failed because: #{self.message}"
      # returning content directly because further constructions are not required
      return content
    when NotificationActionType.id_for(:notify_auto_reload_card_success)
      content = "Auto reload of your expense card with amount INR #{self.amount.value} was successfull"
      # returning content directly because further constructions are not required
      return content
    when NotificationActionType.id_for(:notify_quickbooks_sync_success)
      content = "Syncing of reports with Quickbooks was successful"
      # returning content directly because further constructions are not required
      return content
    when NotificationActionType.id_for(:notify_quickbooks_sync_failure)
      content = "Syncing of reports with Quickbooks was partially successful or failed because : #{self.message}"
      # returning content directly because further constructions are not required
      return content
    when NotificationActionType.id_for(:notify_quickbooks_sync_partial_success)
      content = "Syncing with Quickbooks was partially successful - refer to reports list to see what all were successful"
      return content
    when NotificationActionType.id_for(:notify_credit_confirmation)
      bank_payment_request = BankPaymentRequest.unscoped.find_by(id: self.object_id)
      content = "Credit for one of your payments in Payment Order: #{bank_payment_request.name} was successful!"
      return content
    when NotificationActionType.id_for(:notify_credit_failure)
      # todo
    else
      action = ""
  end

  # adding action
  content += action + object_prep

  amount_value = Amount.find_by(id: self.amount_id).value rescue nil
  if amount_value
    amount = " INR " + amount_value.to_s
  end

  # adding amount
  content += amount if amount_value

  object = ""
  object_name = ""
  case object_type
    when "Report"
      if action_type_id == NotificationActionType.id_for(:notify_credit_reimbursement) || action_type_id == NotificationActionType.id_for(:notify_cash_withdrawal_against_report_reimbursement)
        object = " for your report"
      else
        object = " report"
      end
      r = Report.unscoped.find_by(id: self.object_id)
      object_name = " : \"#{r.name}\""
      if action_type_id == NotificationActionType.id_for(:notify_auto_approve)
        object_name += " has been auto approved"
      end
      if action_type_id == NotificationActionType.id_for(:notify_awaiting_approval)
        object_name += " and is awaiting your further approval"
      end
    when "Money Request"
      if action_type_id == NotificationActionType.id_for(:notify_credit_advance_request)
        object = " for a money request"
        mr = MoneyRequest.unscoped.find_by(id: self.object_id)
        object_name = " : \"#{mr.name}\""
      else
        object = " money request"
        mr = MoneyRequest.unscoped.find_by(id: self.object_id)
        object_name = " : \"#{mr.name}\""
      end
      if action_type_id == NotificationActionType.id_for(:notify_auto_approve)
        object_name += " has been auto approved"
      end
      if action_type_id == NotificationActionType.id_for(:notify_awaiting_approval)
        object_name += " and is awaiting your further approval"
      end
    when "PaymentOrder", "Payment Order"
      if action_type_id == NotificationActionType.id_for(:notify_credit)
        object = " against your payment order"
      else
        object = " payment order"
      end
      bpr = BankPaymentRequest.unscoped.find_by(id: self.object_id)
      object_name = " : \"#{bpr.name}\""
      if action_type_id == NotificationActionType.id_for(:notify_auto_approve)
        object_name += " has been auto approved"
      end
      if action_type_id == NotificationActionType.id_for(:notify_awaiting_approval)
        object_name += " and is awaiting your further approval"
      end
    when "PaymentOrderItem"
      # this is when an individual payment order item is rejected - will hold its details
       # todo - add logic to handle this
    when "Expense"
      object = " an expense"
      e = Expense.unscoped.find_by(id: self.object_id)
      object_name = " at #{e.merchant_name}"
    when "Card"
      object = " to your card"
      # adding conditions for auto reload success and failure

    when "Wallet"
      if action_type_id == NotificationActionType.id_for(:notify_credit)
        object = " to your Finly account"
      else
        object = " from your Finly account"
      end
    when "Payment"
      case action_type_id
        when NotificationActionType.id_for(:notify_cash_load)
          object = " to your cash in hand"
        when NotificationActionType.id_for(:notify_cash_withdrawal)
          object = " from your cash in hand"
        else
          # do nothing
      end
    when "Purchase Order"
      case action_type_id
        when NotificationActionType.id_for(:notify_submit)
          object = " purchase order"
        else
          object = " purchase order"
          # do nothing
      end
    when "Invoice"
      case action_type_id
        when NotificationActionType.id_for(:notify_submit)
          object = " invoice"
        else
          object = " invoice"
          # do nothing
      end
    else
      # do nothing
  end

  # adding object and corresponding name
  content += object + object_name
  return content

end

def generate_link(user)

  # base = Constants::BASE_URL
  view = ""
  path = ""
  suffix = ""

  if ["em", :em].include?(RequestStore.store[:current_app])
    case view_role_id
      when 3
        view = "\"manager\""
      when 2
        view = "\"finance_manager\""
      when 1
        view = "\"admin\""
      else
        view = "\"personal\""
    end
    suffix = "/params={\"view\":" + view + ",\"preview\":true"
    path = ""
  end

  case self.path_object_type
    when "Report"
      path_object = Report.unscoped.find_by(id: self.path_object_id)
      if path_object.is_deleted?
        path = "reports"
      else
        path = "reports/#{self.path_object_id}"
      end
    when "Reports"
      suffix = "/params={\"view\":" + view
      path = "reports"
    when "Money Request"
      path_object = MoneyRequest.unscoped.find_by(id: self.path_object_id)
      if path_object.is_deleted?
        path = "requests"
      else
        path = "requests/#{self.path_object_id}"
      end
    when "Settlement", "Wallet", "Payment", "Card"
      path = "settlements/"
      if view == "\"personal\"" or view.nil?
        path += "#{user.id}"
      end
      suffix += ",\"view_to_load\":\"payments\""
    when "Card"
      if view == 'personal' or view.nil?
        path = "settings"
      else
        path = "employees"
      end
    when "User"
      if view == 'personal' or view.nil?
        # do nothing
      else
        path = "employees"
      end
    when "PaymentOrder", "Payment Order"
      path_object = BankPaymentRequest.unscoped.find_by(id: self.path_object_id)
      if path_object.blank?
        path = "orders"
      else
        path = "orders/#{self.path_object_id}"
      end
    when "Purchase Order"
      document = PurchaseOrder.unscoped.find_by(id: self.path_object_id)
      if document.blank? || document&.procurement.blank?
        path = "purchase_orders"
        suffix = "/params={"
      else
        path = "procure_purchase_order/#{document.id}"
        suffix = "/params={\"procurement_id\":#{document.procurement_id}"
      end
    when 'Invoice'
      document = Invoice.unscoped.find_by(id: self.path_object_id)
      if document.blank? || document&.procurement.blank?
        path = "new_payment_invoices"
        suffix = "/params={"
      else
        path = "procure_invoice/#{document.id}"
        if document.invoice_type_id == InwardInvoiceType.id_for(:tax_invoice)
          suffix = "/params={\"procurement_id\":#{document.procurement_id }"
        elsif document.invoice_type_id == InwardInvoiceType.id_for(:non_po_invoice)
          suffix = "/params={\"procurement_id\":#{document.procurement_id },\"procurement_type\":\"non_po\""
        elsif document.invoice_type_id == InwardInvoiceType.id_for(:agreement_invoice)
          suffix = "/params={\"procurement_id\":#{document.procurement_id },\"procurement_type\":\"non_po\",\"invoice_type\":\"recurring_invoice\""
        end
      end
    else
      # do nothing
  end
  suffix += "}"

  return "#" + path + suffix

end

def retrieve_object_preposition
  case object_type
    when "Report"
      object = Report.unscoped.find_by(id: object_id)
    when "Money Request"
      object = MoneyRequest.unscoped.find_by(id: object_id)
    when 'Purchase Order'
      object = PurchaseOrder.unscoped.find_by(id: object_id)
    when 'Invoice'
      object = Invoice.unscoped.find_by(id: object_id)
    else
      return ""
  end
  owner = object.user rescue (object.creator rescue (object.created_by rescue (object.owner rescue self.user)))
  if self.action_type_id == NotificationActionType.id_for(:notify_auto_approve)
    return ""
  elsif self.action_type_id == NotificationActionType.id_for(:notify_submit)
    return " a" if object_type == "Report"
    return " an" if object_type == "Money Request"
    return " a" if object_type == "Purchase Order"
    return " an" if object_type == "Invoice"
  elsif self.user_id == owner.id
    return " your"
  else
    return " " + owner.name + "'s"
  end
end
