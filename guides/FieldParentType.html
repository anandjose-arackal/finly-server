def self.ids_for(*list)

    ids << TYPES[name]
  end
  ids.uniq.reject {|id| id.nil?}
end

def self.seed_data
  c = self.count
  TYPES.each do |name, id|
    if c == 0 or self.with_name(name).blank?
      self.create name: name
    end
  end
end
