def self.update_resource(options)
  ids = options[:ids]
  resource_id = options[:resource_id]
  user = options[:user]
  if ids.present? && resource_id.present?
    attachments = user.attachments.with_ids(ids).update_all(resource_id: resource_id)
  end
end
