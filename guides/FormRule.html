def fetch_form_data(data, path)
  paths = path.split("/")
  result = data
  while (p = paths.shift)
    return nil if result.nil?
    # Final node
    return result[p] if paths.blank?
    return nil unless result.class <= Hash
    result = result[p]
  end

  nil
end
